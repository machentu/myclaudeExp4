curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg

echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

##########################################################
docker commit 制作镜像步骤：

docker run -it <基础镜像> /bin/bash

在容器内进行修改：

exit


docker commit -m "提交信息" -a "作者" <容器ID或名称> <新镜像名称>:<标签>

常用参数：
-m: 提交的描述信息。
-a: 指定镜像作者。 

注意事项与对比：
不推荐用于生产： docker commit 制作的镜像不可复现，且难以维护，应优先使用 Dockerfile` 构建。
对比 build： docker commit 是手动修改后封装，docker build 是通过脚本自动化生成，更适合版本控制。 

# 假设有一个容器ID为 a1b2c3d4e5f6
docker commit -m "install nginx" -a "user" a1b2c3d4e5f6 my-nginx:v1

docker save -o <保存的文件名>.tar <镜像名>:<标签>


进入正在运行的镜像
docker exec -it <容器ID或名称> /bin/bash
